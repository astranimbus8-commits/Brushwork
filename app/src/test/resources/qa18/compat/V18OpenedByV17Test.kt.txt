// v1.8 QA (compat), kept as text so it never runs again. Run ONCE on the v1.7.0 sources (tag
// v1.7.0, 46837d75). It opens the project v1.8 saved (qa18/compat/v18-written: the v1.8 QA's
// Qa18CompatArtwork, id qa18-compat-v18) with v1.7's ProjectRepository and editor, and writes
// what v1.7 shows and keeps:
// - v17-untouched: opened in the editor, nothing edited, saved (as on leaving).
// - v17-edited: a path added to the vector layers "Mirrored", "Generated" and "Cogs"; the shape
//   layers "Mirrored shape", "Cog" and "Holed cog" turned 10 degrees with the Transform tool; the
//   four arrays' count raised by one; saved.
// Each with probe.tsv: the composite and, per layer, its stored pixels, what v1.7 draws from the
// layer's data ("fresh"), and how v1.7 reads the data (CRC32 of the ARGB ints, big-endian); and
// composite.png. plainShapeTurned is the control: a shape layer v1.7 made itself, turned the same
// way, stores exactly what v1.7 draws from its data; "diff" prints where the edited layers' stored
// pixels differ from it (the pixels v1.8 drew that v1.7 never redrew). Every file of v17-untouched
// but project.json was byte-identical to v18-written (cmp), so only its project.json is kept.

@RunWith(RobolectricTestRunner::class)
class V18OpenedByV17Test {
    private val app get() = RuntimeEnvironment.getApplication()
    private val exRoot = File(System.getenv("TEMP"), "v18qa_compat_exchange")
    private val id = "qa18-compat-v18"

    private fun pixels(b: Bitmap): IntArray = IntArray(b.width * b.height).also { b.getPixels(it, 0, b.width, 0, 0, b.width, b.height) }

    private fun crc(p: IntArray): Long {
        val buf = ByteBuffer.allocate(p.size * 4)
        buf.asIntBuffer().put(p)
        return CRC32().apply { update(buf.array()) }.value
    }

    /** What v1.7 draws from [l]'s data (null: raster, nothing to draw from). */
    private fun fresh(doc: Document, l: Layer): IntArray? {
        val d = l.dataSnapshot()
        val area = Rect(0, 0, doc.width, doc.height)
        val out = BitmapUtils.createLayerBitmap(doc.width, doc.height)
        try {
            val cv = Canvas(out)
            val a = d.array
            when {
                a != null && a.pixels != null -> ArrayDraw.drawPixels(cv, a)
                d.vector != null -> VectorLayerRenderer.render(cv, ArrayDraw.effectiveVector(d)!!, area, tips = TipCache(), document = area)
                d.text != null && a == null -> {
                    val item = TextCodec.decode(d.text)!!
                    TextRenderer.drawItem(cv, item, TextRenderer.prepare(item), null)
                }
                d.text != null || d.shape != null -> {
                    val draw = ArraySources.sourceDraw(d, doc.colorMode, doc.width, doc.height)!!
                    ArrayDraw.drawWithArray(cv, a, ArrayDraw.sourceBounds(d) ?: RectF(), draw)
                }
                else -> return null
            }
            return pixels(out)
        } finally {
            out.recycle()
        }
    }

    private fun install(): File {
        val dir = File(app.filesDir, "projects/$id")
        dir.deleteRecursively()
        dir.mkdirs()
        val src = File(exRoot, "v18-written")
        for (f in src.listFiles()!!) if (f.name != "v18probe.tsv") f.copyTo(File(dir, f.name), overwrite = true)
        return dir
    }

    private fun settle(c: EditorController) {
        c.vectors.flushPending()
        Qa17CompatArtwork.settle(c)
        Smoke.pumpUntil { c.doc.layers.none { c.arrayRenders.isPendingOn(it) } }
        Smoke.pump(40)
    }

    /** How v1.7 reads [l]'s data. */
    private fun describe(l: Layer): String {
        val d = l.dataSnapshot()
        val parts = ArrayList<String>()
        d.array?.let { parts += "array=" + ArrayCodec.encodeSpec(it.spec) }
        d.vector?.let { v ->
            parts += "objects=" + v.objects.joinToString(",") { o ->
                when (o) {
                    is VShape -> "VShape:${o.shape.type}:points=${o.shape.points?.size ?: 0}"
                    is VPath -> "VPath:subpaths=${o.subpaths.size}:spline=${o.spline != null}"
                    else -> o.javaClass.simpleName
                }
            }
        }
        d.shape?.let { s -> ShapeCodec.decode(s)?.let { parts += "shape=${it.type}:points=${it.points?.size ?: 0}" } }
        d.text?.let { parts += "text" }
        return parts.joinToString(";").ifEmpty { "raster" }
    }

    private fun write(doc: Document, dir: File, name: String) {
        val ex = File(exRoot, name)
        ex.deleteRecursively()
        ex.mkdirs()
        val names = dir.list()!!.filter { it != ProjectFormat.THUMB_FILE }.sorted()
        for (n in names) File(dir, n).copyTo(File(ex, n), overwrite = true)
        val loaded = runBlocking { ProjectRepository(app).load(id) }
        val sb = StringBuilder()
        sb.append("warnings\t").append(doc.loadWarnings).append('\n')
        sb.append("files\t").append(names.joinToString(",")).append('\n')
        val flat = Compositor(loaded) { null }.renderFlattened()
        sb.append("composite\t").append(crc(pixels(flat))).append('\n')
        File(ex, "composite.png").outputStream().use { flat.compress(Bitmap.CompressFormat.PNG, 100, it) }
        flat.recycle()
        for (l in loaded.layers) {
            val f = fresh(loaded, l)?.let { crc(it) }?.toString() ?: "-"
            sb.append("layer\t${l.id}\t${l.name}\t${crc(pixels(l.bitmap))}\t$f\t${describe(l)}\n")
        }
        File(ex, "probe.tsv").writeText(sb.toString())
        println(sb)
    }

    @Test
    fun untouched() {
        val dir = install()
        val doc = runBlocking { ProjectRepository(app).load(id) }
        assertEquals(emptyList<String>(), doc.loadWarnings)
        val c = Smoke.controller(app, doc)
        c.viewTransform.set(Matrix())
        settle(c)
        assertEquals("v1.7 edits nothing on opening", 0, c.editCount)
        runBlocking { ProjectRepository(app).save(doc, null) }
        c.dispose()
        write(doc, dir, "v17-untouched")
    }

    private fun turn(c: EditorController, layer: Layer) {
        c.selectLayer(layer)
        c.selectTool(ToolId.TRANSFORM)
        Smoke.pump(20)
        val tool = c.tools.getValue(ToolId.TRANSFORM) as TransformTool
        tool.snapToObjects = false
        tool.start()
        assertEquals("${layer.name}: kept as data", TransformTool.Lifted.SHAPE, tool.lifted)
        tool.setRotation(10.0)
        tool.endNumericEdit()
        tool.commit()
        c.selectTool(ToolId.BRUSH)
        settle(c)
    }

    @Test
    fun plainShapeTurned() {
        val doc = Smoke.document(400, 300, layers = 1, whiteBottom = true)
        val c = Smoke.controller(app, doc)
        c.viewTransform.set(Matrix())
        c.snapping.enabled = false
        val o = com.brushwork.paint.tools.vector.ShapeObject(com.brushwork.paint.tools.vector.ShapeType.ELLIPSE, cx = 150f, cy = 200f, w = 40f, h = 20f, rotation = 20f, style = com.brushwork.paint.tools.vector.ShapeStyle.STROKE, strokeWidth = 4f, strokeColor = 0xFF2266AA.toInt())
        val l = c.addLayerWithContent("Plain", "Add shape", shapeData = ShapeCodec.encode(o), draw = ArraySources.shapeDraw(o, com.brushwork.paint.model.ColorMode.RGB, 400, 300))!!
        settle(c)
        println("plain before: stored ${crc(pixels(l.bitmap))} fresh ${crc(fresh(doc, l)!!)}")
        turn(c, l)
        println("plain turned: stored ${crc(pixels(l.bitmap))} fresh ${crc(fresh(doc, l)!!)}")
        c.dispose()
    }

    @Test
    fun edited() {
        val dir = install()
        val doc = runBlocking { ProjectRepository(app).load(id) }
        assertEquals(emptyList<String>(), doc.loadWarnings)
        val c = Smoke.controller(app, doc)
        c.viewTransform.set(Matrix())
        c.snapping.enabled = false
        settle(c)
        fun layer(n: String) = doc.layers.single { it.name == n }
        val dot = VPath(0, subpaths = listOf(VSubpath(listOf(VAnchor(196f, 146f), VAnchor(204f, 154f)))), stroke = VStrokeStyle(color = 0xFF000000.toInt(), width = 2f))
        for (n in listOf("Mirrored", "Generated", "Cogs")) {
            c.selectLayer(layer(n))
            assertTrue(n, c.vectors.addObjects(layer(n), listOf(dot), "Add").isNotEmpty())
            settle(c)
        }
        for (n in listOf("Mirrored shape", "Cog", "Holed cog")) turn(c, layer(n))
        for (n in listOf("Mirrored", "Generated", "Cogs", "Mirrored shape", "Cog", "Holed cog")) {
            val s = pixels(layer(n).bitmap); val f = fresh(doc, layer(n))!!
            var cnt = 0; var x0 = 9999; var y0 = 9999; var x1 = -1; var y1 = -1
            for (i in s.indices) if (s[i] != f[i]) { cnt++; val x = i % 400; val y = i / 400; x0 = minOf(x0, x); y0 = minOf(y0, y); x1 = maxOf(x1, x); y1 = maxOf(y1, y) }
            println("diff $n: $cnt px in [$x0,$y0..$x1,$y1]")
        }
        for (n in listOf("Array 1", "Text array", "Shape array", "Vector array")) {
            val l = layer(n)
            assertTrue(n, ArrayOps.edit(c, l, l.array!!.spec.copy(count = l.array!!.spec.count + 1)))
            settle(c)
        }
        assertTrue(c.editCount > 0)
        runBlocking { ProjectRepository(app).save(doc, null) }
        c.dispose()
        write(doc, dir, "v17-edited")
    }
}
